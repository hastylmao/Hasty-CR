### 1.1 Dynamic Tower HP Bar Localization
**Root Cause**: HP bar vertical position varies ±34 pixels depending on match type (1v1, 2v2, special event) and UI scaling. Fixed ROI (y=1160-1177) fails.

**Solution**: Multi-stage landmark detection with adaptive ROI:

```python
class DynamicHPDetector:
    def __init__(self):
        self.landmark_templates = {
            'ally_crown_tower': cv2.imread('templates/ally_crown.png'),
            'enemy_crown_tower': cv2.imread('templates/enemy_crown.png'),
            'level_badge': cv2.imread('templates/level_badge.png')
        }
        
    def locate_tower_hp(self, frame):
        # Stage 1: Crown tower detection via template matching
        crown_matches = self._template_match(frame, 'ally_crown_tower')
        if len(crown_matches) < 2:
            # Stage 2: Fallback to level badge detection
            badge_matches = self._template_match(frame, 'level_badge')
            if len(badge_matches) >= 2:
                return self._infer_hp_bars_from_badges(badge_matches)
            # Stage 3: Color segmentation for HP bar green/red
            return self._color_segmentation_fallback(frame)
        
        # Anchor HP bars relative to crown positions
        crown_y = np.mean([match['y'] for match in crown_matches])
        hp_roi_y = crown_y + 85  # Pixel offset from crown to HP bar
        return (int(hp_roi_y), int(hp_roi_y + 17))  # 17px bar height
    
    def _template_match(self, frame, template_key):
        """Multi-scale template matching with confidence thresholding"""
        template = self.landmark_templates[template_key]
        result = cv2.matchTemplate(frame, template, cv2.TM_CCOEFF_NORMED)
        locations = np.where(result >= 0.85)
        return [{'x': x, 'y': y} for y, x in zip(*locations)]
    
    def _color_segmentation_fallback(self, frame):
        """HP bars have unique HSV ranges: Green(ally): H60-90, Red(enemy): H0-10|170-180"""
        hsv = cv2.cvtColor(frame, cv2.COLOR_BGR2HSV)
        
        # Green HP bar (ally)
        green_lower = np.array([60, 100, 100])
        green_upper = np.array([90, 255, 255])
        green_mask = cv2.inRange(hsv, green_lower, green_upper)
        
        # Find largest green contour in top half
        contours, _ = cv2.findContours(green_mask, cv2.RETR_EXTERNAL, 
                                       cv2.CHAIN_APPROX_SIMPLE)
        if contours:
            largest = max(contours, key=cv2.contourArea)
            y_pos = cv2.boundingRect(largest)[1]
            return (y_pos, y_pos + 17)
        
        return (1160, 1177)  # Ultimate fallback
```

class ActionQueueManager:
    def __init__(self):
        self.queue = deque(maxlen=3)
        self.lock = threading.Lock()
        
    def submit_action(self, action, confidence=0.0):
        """Submit action with confidence score"""
        with self.lock:
            self.queue.append({
                'action': action,
                'timestamp': time.time(),
                'confidence': confidence,
                'prevalidated': False
            })
            
    def execute_next(self, current_state):
        """Execute next action if still valid"""
        if not self.queue:
            return False
            
        next_action = self.queue[0]
        
        # Revalidate against current state
        if self._validate_action(next_action['action'], current_state):
            execute_action(next_action['action'])
            self.queue.popleft()
            return True
        else:
            # Action invalidated by state change
            self.queue.popleft()
            return False
    
    def _validate_action(self, action, state):
        """Check if action prerequisites still hold"""
        card_slot, target_x, target_y, delay = action
        
        # Check card still in expected slot
        expected_card = self._get_expected_card(card_slot)
        actual_card = state['hand'][card_slot]
        
        if expected_card != actual_card:
            return False
        
        # Check elixir sufficient
        card_cost = self._card_cost(actual_card)
        if state['elixir'] < card_cost:
            return False
            
        # Check placement area not blocked
        if not self._placement_available(target_x, target_y, state):
            return False
            
        return True

        Core principle: Kite to opposite lane, Cannon pocket placement
        """
        # Identify main tank
        main_tank = self._find_main_tank(threats)
        
        if main_tank['type'] in ['GOLEM', 'LAVA_HOUND']:
            # Heavy beatdown: Full opposite lane kite

                    unit_speed=main_tank['speed']
                )
                
                # Place Ice Golem at calculated intercept point
                intercept_tile = kite_tiles[2]  # Third tile in path
                return ('Ice Golem', intercept_tile[0], intercept_tile[1], 0)
            
            # Cannon placement: 4-3 or 2-3 pocket
            if 'Cannon' in hand and elixir >= 3:
                pocket_tile = self._select_cannon_pocket(main_tank['lane'])
                return ('Cannon', pocket_tile[0], pocket_tile[1], 100)
                
        elif main_tank['type'] == 'ROYAL_GIANT':
            # Royal Giant defense: Cannon at river edge
            rg_tile = main_tank['tile']
            if rg_tile[1] >= 16:  # Enemy side
                # RG targets buildings from distance
                cannon_x = 8 if rg_tile[0] < 9 else 10  # Left or right pocket
                cannon_y = 21  # One tile forward from king
                return ('Cannon', cannon_x, cannon_y, 0)
        
        return None  # No optimal defense found
    
    def _compute_kite_path(self, start_tile, target_lane, unit_speed):
        """
        Calculate optimal kite path using vector fields
        """
        # Create attraction vector to target lane
        target_x = 9 if target_lane == 'LEFT' else 13
        target_y = 28  # Deep in opposite lane
        
        # Path calculation using potential fields
        path = []
        current = start_tile
        
        for step in range(10):  # Max 10 tiles
            # Calculate gradient toward target
            dx = target_x - current[0]
            dy = target_y - current[1]
            
            # Normalize and apply unit speed
            norm = np.sqrt(dx**2 + dy**2)
            if norm > 0:
                dx, dy = dx/norm, dy/norm
            
            next_tile = (
                current[0] + dx * unit_speed,
                current[1] + dy * unit_speed
            )
            
            # Clamp to valid tiles
            next_tile = (
                max(0, min(17, int(next_tile[0]))),
                max(0, min(31, int(next_tile[1])))
            )
            
            path.append(next_tile)
            current = next_tile
            
            if current[1] >= 28:  # Reached deep opposite lane
                break
        
        return path

            # 4-3 placement: x=8, y=21
            return (8, 21)
        else:

            return (10, 21)
