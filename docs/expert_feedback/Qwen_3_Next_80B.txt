#### **Step 2: Compute Affine Transform Matrix**
Let:
- `P_ref = [KingTower, LeftPrincess, RightPrincess, ElixirStart]` — reference coordinates from calibration match
- `P_curr = [KingTower’, LeftPrincess’, RightPrincess’, ElixirStart’]` — current match

Compute **affine transform matrix** `M` (3x3) mapping `P_ref → P_curr` via **Least Squares**:

```python
import numpy as np

def compute_affine_matrix(P_ref, P_curr):
    # P_ref, P_curr: 4x2 arrays (x, y)
    A = np.zeros((8, 6))
    b = np.zeros((8, 1))
    
    for i in range(4):
        x, y = P_ref[i]
        x’, y’ = P_curr[i]
        A[2*i]   = [x, y, 1, 0, 0, 0]
        A[2*i+1] = [0, 0, 0, x, y, 1]
        b[2*i]   = x’
        b[2*i+1] = y’

    # Solve: A * [a, b, c, d, e, f]^T = b
    coeffs, residuals, rank, s = np.linalg.lstsq(A, b, rcond=None)
    M = np.array([[coeffs[0], coeffs[1], coeffs[2]],
                  [coeffs[3], coeffs[4], coeffs[5]],
                  [0,         0,         1]])
    return M
```

```python
def estimate_hp_from_profile(profile, max_width=260):
    # profile: 1D array of red channel intensities
    # Find start: first index where red > 200 and green+blue < 80
    start = None
    for i in range(len(profile)):
        if profile[i] > 200 and profile[i-1] < 50:  # assume drop-off at edge
            start = i
            break
    if not start: return 0.0

    # Find end: last pixel where red > 100
    end = start
    for i in range(start, len(profile)):
        if profile[i] < 100:
            break
        end = i

    hp_ratio = (end - start + 1) / max_width
    return max(0.0, min(1.0, hp_ratio))
```

## **1.2 Unit Detection Fidelity — Handling Overlapping Swarms**

### Problem:
Clash Royale units (Skeletons, Ice Spirits) spawn in swarms. Bounding boxes overlap → ONNX model outputs 5–6 boxes for 10 units → low precision.

### Solution: **Multi-Scale Clustering + Density-Based Instance Segmentation**

#### **Step 1: Detect All Bounding Boxes**

#### **Step 2: Group by Class and Scale**
```python
from sklearn.cluster import DBSCAN
from scipy.spatial.distance import cdist

def cluster_units(boxes, class_id, eps=15, min_samples=2):
    # Extract centers
    centers = np.array([( (x1+x2)/2, (y1+y2)/2 ) for x1,y1,x2,y2,_,_ in boxes if class_id == _])
    if len(centers) == 0: return []

    # DBSCAN on center points
    clustering = DBSCAN(eps=eps, min_samples=min_samples).fit(centers)
    labels = clustering.labels_

    clusters = []
    for label in np.unique(labels):
        if label == -1: continue  # noise
        cluster_boxes = [boxes[i] for i in range(len(boxes)) if labels[i] == label]
        # Merge boxes: take union
        x1 = min(b[0] for b in cluster_boxes)
        y1 = min(b[1] for b in cluster_boxes)
        x2 = max(b[2] for b in cluster_boxes)
        y2 = max(b[3] for b in cluster_boxes)
        count = len(cluster_boxes)
        avg_conf = np.mean([b[5] for b in cluster_boxes])
        clusters.append((x1,y1,x2,y2, class_id, avg_conf, count))
    return clusters
```

#### **Step 3: Estimate Unit Count via Density**
- Skeletons → 1 pixel per unit ≈ 36px²/unit → area / 36 ≈ count
- Ice Spirits → 40px²/unit
- Musketeer → 180px²/unit

```python
def estimate_unit_count(box, class_id):
    area = (box[2] - box[0]) * (box[3] - box[1])
    densities = {
        'skeleton': 36,
        'ice_spirit': 40,
        'ice_golem': 220,
        'musketeer': 180,
        'hog': 200
    }
    return max(1, int(area / densities.get(class_id, 100)))
```

```python
def minicap_stream():
    sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    sock.connect(('127.0.0.1', 1313))  # minicap default port
    header = sock.recv(24)
    width, height = struct.unpack('II', header[4:12])
    while True:
        # Read frame length
        length_bytes = sock.recv(4)
        if len(length_bytes) < 4: break
        frame_len = struct.unpack('I', length_bytes)[0]
        frame_data = b""
```

```python
def predict_state(s_t, delay_ms=1000):
    # Elixir: 1 unit per 2.8s → 0.357 per second
    elixir_pred = min(10, s_t['elixir'] + (delay_ms / 1000) * 0.357)
    
    # Card cycle: 2.6s per card (Hog 2.6)
    # hand_cards: [c0, c1, c2, c3, c4] — c4 is queued
    # Assume cards are consumed in order
    card_consumption_rate = 2.6  # seconds per card
    
    # How many cards consumed in delay_ms?
    cards_consumed = delay_ms / (card_consumption_rate * 1000)
    
    # Simulate card shift
    hand_new = s_t['hand_cards'][:]
    for _ in range(int(cards_consumed)):
        hand_new = hand_new[1:] + [hand_new[0]]  # rotate
    # Add probabilistic refill based on cycle
    if np.random.rand() < 0.8:  # 80% chance card refilled
        hand_new[4] = s_t['prev_card']  # last used card is next queued

    return {
        'elixir': elixir_pred,
        'hand_cards': hand_new,
        'tower_hp': s_t['tower_hp'],  # assume static
        'units': s_t['units'],
        'time': s_t['time'] + delay_ms/1000
    }
```
